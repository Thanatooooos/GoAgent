**Files:**

- [ ] **Step 1: wikiService.ts**（按设计 §二）

- [ ] **Step 2: wikiLinks.ts** — `convertWikiLinks(content: string, kbId: string): string`：把 `[[slug|标题]]` 替换为 `[标题](/admin/knowledge/${kbId}/wiki?slug=${encodeURIComponent(slug)})`；`[[slug]]` 无标题时用 slug。

```ts
export function convertWikiLinks(content: string, kbId: string): string {
  return content.replace(/\[\[([^\]|]+)(?:\|([^\]]*))?\]\]/g, (_m, slug, title) => {
    const s = String(slug).trim();
    const t = (title || slug).trim();
    if (!s) return _m;
    return `[${t}](/admin/knowledge/${encodeURIComponent(kbId)}/wiki?slug=${encodeURIComponent(s)})`;
  });
}
```
